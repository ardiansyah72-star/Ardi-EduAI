/* =====================================================
   FUNGSI MENGAMBIL FITUR
   ===================================================== */

function getSelectedFeatures() {

    let select =
        document.getElementById("fitur");

    let selected = [];

    for (let option of select.options) {

        if (option.selected) {

            selected.push(option.value);

        }

    }

    if (selected.length === 0) {

        return ["Skor", "Feedback"];

    }

    return selected;
}

/* =====================================================
   SALIN PROMPT
   ===================================================== */

function copyPrompt() {

    let prompt =
        document.getElementById("hasilPrompt").value;

    if (prompt.trim() === "") {

        alert("Belum ada prompt yang bisa disalin.");

        return;

    }


    if (navigator.clipboard) {

        navigator.clipboard.writeText(prompt)
            .then(function() {

                alert("✅ Prompt berhasil disalin!");

            })
            .catch(function() {

                fallbackCopy();

            });

    } else {

        fallbackCopy();

    }

}


function fallbackCopy() {

    let textarea =
        document.getElementById("hasilPrompt");

    textarea.select();

    document.execCommand("copy");

    alert("✅ Prompt berhasil disalin!");

}


/* =====================================================
   SIMPAN PROMPT
   ===================================================== */

function savePrompt() {

    let prompt =
        document.getElementById("hasilPrompt").value;

    if (prompt.trim() === "") {

        alert("Buat prompt terlebih dahulu.");

        return;

    }


    let history =
        JSON.parse(
            localStorage.getItem("promptHistory") || "[]"
        );


    let item = {

        id: Date.now(),

        title:
            document.getElementById("materi").value ||
            "Prompt Tanpa Judul",

        mapel:
            document.getElementById("mapel").value,

        kelas:
            document.getElementById("kelas").value,

        media:
            document.getElementById("media").value,

        prompt: prompt,

        favorite: false,

        date:
            new Date().toLocaleString("id-ID")

    };


    history.unshift(item);


    /* Maksimal 50 riwayat */

    if (history.length > 50) {

        history = history.slice(0, 50);

    }


    localStorage.setItem(
        "promptHistory",
        JSON.stringify(history)
    );


    alert("💾 Prompt berhasil disimpan!");

    displayHistory();

    updateStatistics();

}

function favoritePrompt() {

        alert("Buat prompt terlebih dahulu.");

    }

            localStorage.getItem("promptHistory") || "[]"
        );


    let item = {

        id: Date.now(),

            document.getElementById("materi").value ||
            "Prompt Favorit",

        mapel:

        media:

        history = history.slice(0, 50);

    }


    localStorage.setItem(
        "promptHistory",
        JSON.stringify(history)
    );


    alert("⭐ Prompt ditambahkan ke favorit!");

    displayHistory();

    updateStatistics();

}


/* =====================================================
   TAMPILKAN RIWAYAT
   ===================================================== */

function displayHistory() {

    let list =
        document.getElementById("historyList");

    let history =
        JSON.parse(
            localStorage.getItem("promptHistory") || "[]"
        );


    let search =
        document.getElementById("searchHistory")
            .value
            .toLowerCase();


    let filtered =
        history.filter(function(item) {

            return (

                item.title.toLowerCase().includes(search) ||

                item.mapel.toLowerCase().includes(search) ||

                item.media.toLowerCase().includes(search)

            );

        });


    if (filtered.length === 0) {

        list.innerHTML = `
            <div class="empty">
                📭 Belum ada riwayat prompt.
            </div>
        `;

        return;

    }


    list.innerHTML = "";


    filtered.forEach(function(item) {

        let div =
            document.createElement("div");

        div.className =
            "history-item";


        div.innerHTML = `

            <h4>
                ${item.favorite ? "⭐ " : ""}
                ${escapeHtml(item.title)}
            </h4>

            <div class="history-meta">

                ${escapeHtml(item.mapel)}
                |
                ${escapeHtml(item.kelas)}
                |
                ${escapeHtml(item.media)}

                <br>

                ${escapeHtml(item.date)}

            </div>

            <div class="history-actions">

                <button
                    class="btn-copy btn-small"
                    onclick="loadPrompt(${item.id})"
                >
                    📂 Buka
                </button>

                <button
                    class="btn-clear btn-small"
                    onclick="deletePrompt(${item.id})"
                >
                    🗑️ Hapus
                </button>

            </div>
        `;


        list.appendChild(div);

    });

}


/* =====================================================
   MEMBUKA RIWAYAT
   ===================================================== */

function loadPrompt(id) {

    let history =
        JSON.parse(
            localStorage.getItem("promptHistory") || "[]"
        );


    let item =
        history.find(function(item) {

            return item.id === id;

        });


    if (!item) {

        return;

    }


    document.getElementById("hasilPrompt").value =
        item.prompt;


    document.getElementById("mapel").value =
        item.mapel;

    document.getElementById("kelas").value =
        item.kelas;

    document.getElementById("media").value =
        item.media;

    document.getElementById("materi").value =
        item.title;


    window.scrollTo({

        top: 0,

        behavior: "smooth"

    });

}


/* =====================================================
   HAPUS RIWAYAT
   ===================================================== */

function deletePrompt(id) {

    let yakin =
        confirm("Hapus prompt ini dari riwayat?");


    if (!yakin) {

        return;

    }


    let history =
        JSON.parse(
            localStorage.getItem("promptHistory") || "[]"
        );


    history =
        history.filter(function(item) {

            return item.id !== id;

        });


    localStorage.setItem(
        "promptHistory",
        JSON.stringify(history)
    );


    displayHistory();

    updateStatistics();

}


/* =====================================================
   RESET FORM
   ===================================================== */

function resetForm() {

    document.getElementById("mapel").selectedIndex = 0;

    document.getElementById("kelas").selectedIndex = 0;

    document.getElementById("materi").value = "";

    document.getElementById("tujuan").value = "";

    document.getElementById("karakter").selectedIndex = 0;

    document.getElementById("media").selectedIndex = 0;

    document.getElementById("kesulitan").selectedIndex = 0;

    document.getElementById("jumlahSoal").value = 10;


    let fitur =
        document.getElementById("fitur");


    for (let option of fitur.options) {

        option.selected = false;

    }


    document.getElementById("hasilPrompt").value = "";

}


/* =====================================================
   BERSIHKAN HASIL
   ===================================================== */

function clearResult() {

    document.getElementById("hasilPrompt").value = "";

}


/* =====================================================
   STATISTIK
   ===================================================== */

function updateStatistics() {

    let history =
        JSON.parse(
            localStorage.getItem("promptHistory") || "[]"
        );


    let favorites =
        history.filter(function(item) {

            return item.favorite === true;

        });


    let total =
        localStorage.getItem("totalDibuat") || "0";


    document.getElementById("jumlahRiwayat")
        .textContent = history.length;


    document.getElementById("jumlahFavorit")
        .textContent = favorites.length;


    document.getElementById("jumlahDibuat")
        .textContent = total;

}


/* =====================================================
   KEAMANAN TAMPILAN TEKS
   ===================================================== */

function escapeHtml(text) {

    return String(text)

        .replace(/&/g, "&amp;")

        .replace(/</g, "&lt;")

        .replace(/>/g, "&gt;")

        .replace(/"/g, "&quot;")

        .replace(/'/g, "&#039;");

}


/* =====================================================
   MENJALANKAN SAAT HALAMAN DIBUKA
   ===================================================== */

displayHistory();

updateStatistics();
